local paintPartsBackup = {}
local paintConnections = {}
local function deleteAllPaintParts()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") and obj.Name == "PaintPlayerPart" then
            local clone = obj:Clone(); clone.Archivable = true
            paintPartsBackup[obj:GetDebugId()] = { clone = clone, parent = obj.Parent }
            obj:Destroy()
        end
    end
end
local function restorePaintParts()
    for _, data in pairs(paintPartsBackup) do
        if data.clone and data.parent then data.clone.Parent = data.parent end
    end
    paintPartsBackup = {}
end
local function watchNewPaintParts()
    table.insert(paintConnections, Workspace.DescendantAdded:Connect(function(obj)
        if obj:IsA("BasePart") and obj.Name == "PaintPlayerPart" then
            task.defer(function()
                if obj and obj.Parent then
                    local clone = obj:Clone(); clone.Archivable = true
                    paintPartsBackup[obj:GetDebugId()] = { clone = clone, parent = obj.Parent }
                    obj:Destroy()
                end
            end)
        end
    end))
end
local function disconnectWatchers()
    for _, conn in ipairs(paintConnections) do if conn.Connected then conn:Disconnect() end end
    paintConnections = {}
end
local function setTouchQuery(state)
    local char = Workspace:FindFirstChild(Player.Name)

    for _, v in ipairs(char:GetChildren()) do
        if v:IsA("Part") or v:IsA("BasePart") then
            v.CanTouch = state; v.CanQuery = state
        end
    end
end

local function startAntiGucci()
    local character = Player.Character or Player.CharacterAdded:Wait()
    local humanoid  = character:WaitForChild("Humanoid")
    local rootPart  = character:WaitForChild("HumanoidRootPart")
    safePosition = rootPart.Position
    local folder = Workspace:FindFirstChild(Player.Name .. "SpawnedInToys")
    local blob   = folder and folder:FindFirstChild("CreatureBlobman")
    local seat   = blob and blob:FindFirstChild("VehicleSeat")
    if not blob then
        spawnBlobman(); task.wait(1)
        folder = Workspace:FindFirstChild(Player.Name .. "SpawnedInToys")
        blob = folder and folder:FindFirstChild("CreatureBlobman")
        seat = blob and blob:FindFirstChild("VehicleSeat")
    end
    if seat and seat:IsA("VehicleSeat") then
        rootPart.CFrame = seat.CFrame + Vector3.new(0, 2, 0)
        seat:Sit(humanoid)
    end
    humanoid:GetPropertyChangedSignal("Jump"):Connect(function()
        if humanoid.Jump and humanoid.Sit then restoreFrames = 15; safePosition = rootPart.Position end
    end)
    if antiGucciConnection then antiGucciConnection:Disconnect() end
    antiGucciConnection = R.Heartbeat:Connect(function()
        if not rootPart or not humanoid then return end
        ReplicatedStorage.CharacterEvents.RagdollRemote:FireServer(rootPart, 0)
        if restoreFrames > 0 then
            rootPart.CFrame = CFrame.new(safePosition); restoreFrames -= 1
        end
    end)
    task.spawn(function()
        while humanoid.Sit do task.wait(1) end
        task.wait(0.5)
        rootPart.CFrame = CFrame.new(safePosition)
    end)
end

    if antiGucciConnection then antiGucciConnection:Disconnect(); antiGucciConnection = nil end

-- ============================================================================
-- DEFENSE: ANTI BRING V2
-- Scans by CLASS (Weld, WeldConstraint, RigidConstraint, AlignPosition,
-- AlignOrientation, Motor6D) AND by name – catching every possible bring method.
-- Also watches your own character for incoming constraints.
-- ============================================================================
local AntiBringConnection
local AntiBringCharConn

local BRING_NAMES = {
    LeftWeld = true, RightWeld = true,
    LeftAlignOrientation = true, RightAlignOrientation = true,
    BringConstraint = true, CarryWeld = true,
}
local BRING_CLASSES = {
    WeldConstraint = true, RigidConstraint = true,
    AlignPosition  = true, AlignOrientation = true,
}

local function isBringConstraint(obj)
    if not obj or not obj.Parent then return false end
    -- Name-based (existing logic)
    if BRING_NAMES[obj.Name] then return true end
    -- Class-based: generic constraint that appeared on character parts
    if BRING_CLASSES[obj.ClassName] then return true end
    return false
end

local function disableConstraint(obj)
    pcall(function()
        if obj:IsA("BasePart") then obj.Enabled = false end
        -- For WeldConstraint / RigidConstraint
        if obj:FindFirstAncestorOfClass("Model") then
            if obj.ClassName == "WeldConstraint" or obj.ClassName == "RigidConstraint" then
                obj.Enabled = false
            end
            if obj.ClassName == "AlignPosition" or obj.ClassName == "AlignOrientation" then
                obj.Enabled = false
            end
        end
    end)
end

local function scanAndDisableAllBrings(state)
    local myToysName = Player.Name .. "SpawnedInToys"
    for _, x in pairs(workspace:GetDescendants()) do
        if isBringConstraint(x) then
            local topModel = x:FindFirstAncestorOfClass("Model")
            if topModel and topModel.Name ~= myToysName then
                if state then
                    disableConstraint(x)
                else
                    pcall(function() x.Enabled = true end)
                end
            end
        end
    end
    -- Also scan your own character for stray incoming welds
    local char = Player.Character
    if char and state then
        for _, x in pairs(char:GetDescendants()) do
            if x:IsA("WeldConstraint") or x:IsA("RigidConstraint") then
                -- Destroy any weld that references a non-character Part0/Part1
                local p0 = pcall(function() return x.Part0 end) and x.Part0
                local p1 = pcall(function() return x.Part1 end) and x.Part1
                if (p0 and not char:IsAncestorOf(p0)) or (p1 and not char:IsAncestorOf(p1)) then
                    x:Destroy()
                end
            end
        end
    end
end

DefenseGroup:CreateToggle({
    Name     = "Anti Bring V2",
    Flag     = "AntiBringV2",
    Default  = false,
    Info     = "Blocks ALL bring constraints by class & name. Auto-restarts on respawn.",
    Callback = function(v)
        SetToggleState("AntiBringV2", v)
        local myToysName = Player.Name .. "SpawnedInToys"

        if AntiBringConnection then AntiBringConnection:Disconnect(); AntiBringConnection = nil end
        if AntiBringCharConn   then AntiBringCharConn:Disconnect();   AntiBringCharConn   = nil end

        if v then
            scanAndDisableAllBrings(true)

            -- Global workspace listener
            AntiBringConnection = workspace.DescendantAdded:Connect(function(x)
                if isBringConstraint(x) then
                    local topModel = x:FindFirstAncestorOfClass("Model")
                    if topModel and topModel.Name ~= myToysName then
                        -- Immediate disable + deferred destroy for hard-bring attempts
                        pcall(function() x.Enabled = false end)
                        task.defer(function()
                            if x and x.Parent then
                                pcall(function() x.Enabled = false end)
                            end
                        end)
                    end
                end
            end)

            -- Character-level listener: destroys welds added directly to your parts
            local function watchChar(char)
                if AntiBringCharConn then AntiBringCharConn:Disconnect() end
                AntiBringCharConn = char.DescendantAdded:Connect(function(x)
                    if not GetToggleState("AntiBringV2") then return end
                    if x:IsA("WeldConstraint") or x:IsA("RigidConstraint") or x:IsA("AlignPosition") then
                        task.defer(function()
                            if x and x.Parent then x:Destroy() end
                        end)
                    end
                end)
            end

            local char = Player.Character
            if char then watchChar(char) end
            _defenseRestarters["AntiBringV2"] = watchChar
        else
            scanAndDisableAllBrings(false)
            _defenseRestarters["AntiBringV2"] = nil
        end
    end
})

-- ============================================================================
-- DEFENSE: ANTI GRAB ULTRA (Triple-threaded)
-- Runs on Heartbeat + Stepped + RenderStepped simultaneously.
-- Fires Struggle + Ragdoll reset + physics zero every available event.
-- ============================================================================
local _antiGrabUltraConns = {}
local _antiGrabUltraActive = false

local function _stopAntiGrabUltra()
    _antiGrabUltraActive = false
    for _, c in pairs(_antiGrabUltraConns) do if c and c.Connected then c:Disconnect() end end
    _antiGrabUltraConns = {}
    local Root = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
    if Root then Root.Anchored = false end
end

local function _startAntiGrabUltra()
    _stopAntiGrabUltra()
    _antiGrabUltraActive = true

    local isHeld = Player:FindFirstChild("IsHeld")
    if not isHeld then return end

    local function doDefense(char, hrp, hum)
        if not char or not hrp or not hum then return end
        pcall(function()
            StruggleEvent:FireServer()
            ReplicatedStorage.CharacterEvents.RagdollRemote:FireServer(hrp, 0)
            ReplicatedStorage.GameCorrectionEvents.StopAllVelocity:FireServer()
        end)
        pcall(function()
            hum.Sit = false
            hum:ChangeState(Enum.HumanoidStateType.GettingUp)
            hum:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
            hrp.Anchored = true
            hrp.AssemblyLinearVelocity  = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end)
        -- Also kill incoming welds on HRP
        for _, obj in pairs(hrp:GetChildren()) do
            if obj:IsA("WeldConstraint") or obj:IsA("RigidConstraint") or
               obj:IsA("AlignPosition")  or obj:IsA("AlignOrientation") then
                pcall(function() obj:Destroy() end)
            end
        end
    end

    local savedCF = nil

    -- Connection 1: Heartbeat (physics-tick)
    table.insert(_antiGrabUltraConns, R.Heartbeat:Connect(function()
        if not _antiGrabUltraActive or not isHeld.Value then return end
        local char = Player.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        if savedCF and hrp then hrp.CFrame = savedCF end
        doDefense(char, hrp, hum)
    end))

    -- Connection 2: Stepped (before physics solve)
    table.insert(_antiGrabUltraConns, R.Stepped:Connect(function()
        if not _antiGrabUltraActive or not isHeld.Value then return end
        local char = Player.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        doDefense(char, hrp, hum)
    end))

    -- Connection 3: RenderStepped (every render frame = position lock)
    table.insert(_antiGrabUltraConns, R.RenderStepped:Connect(function()
        if not _antiGrabUltraActive or not isHeld.Value then return end
        local char = Player.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        if savedCF then
            hrp.CFrame = savedCF
            hrp.AssemblyLinearVelocity  = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
    end))

    -- IsHeld watcher: capture position on grab, release on drop
    table.insert(_antiGrabUltraConns, isHeld.Changed:Connect(function(val)
        if not _antiGrabUltraActive then return end
        local char = Player.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        if val then
            -- Just got grabbed - freeze position
            if hrp then
                savedCF = hrp.CFrame
                hrp.Anchored = true
                -- Ghost: disable collision on all parts
                for _, p in pairs(char:GetDescendants()) do
                    if p:IsA("BasePart") then p.CanCollide = false end
                end
            end
        else
            -- Released - unfreeze
            if hrp then
                hrp.Anchored = false
                if savedCF then hrp.CFrame = savedCF end
                savedCF = nil
            end
            if char then
                for _, p in pairs(char:GetDescendants()) do
                    if p:IsA("BasePart") then p.CanCollide = true end
                end
            end
        end
    end))

    -- If already held when enabled
    if isHeld.Value then
        local char = Player.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then savedCF = hrp.CFrame; hrp.Anchored = true end
    end
end

    local toysFolder = workspace:FindFirstChild(Player.Name .. "SpawnedInToys")
    if not toysFolder then return {} end
    local whitelist = {}
    for _, toy in pairs(toysFolder:GetChildren()) do
        local nm = toy.Name:lower()
        -- Blobman
        if nm == "creatureblobman" then
            table.insert(whitelist, toy)
        end
        -- Wood plank (various possible names in Fling Things)
        if nm:find("wood") or nm:find("plank") or nm == "planky" or nm == "plankywoodplank" or nm == "woodplanky" then
            table.insert(whitelist, toy)
        end
    end
    return whitelist
end

local function _isPartInWhitelist(part, whitelist)
    for _, model in ipairs(whitelist) do
        if model == part or model:IsAncestorOf(part) then return true end
    end
    return false
end

DefenseGroup:CreateToggle({
    Name    = "No Object Touch",
    Flag    = "NoObjectTouch",
    Default = false,
    Info    = "Blocks ALL objects from touching/affecting you. Exempts YOUR Blobman and YOUR Wood Plank only.",
    Callback = function(v)
        SetToggleState("NoObjectTouch", v)
        _noTouchActive = v

        if _noTouchConn then _noTouchConn:Disconnect(); _noTouchConn = nil end

        if not v then
            -- Restore character CanTouch on disable
            local char = Player.Character
            if char then
                for _, p in pairs(char:GetDescendants()) do
                    if p:IsA("BasePart") then
                        p.CanTouch = true
                        p.CanQuery = true
                    end
                end
            end
            notify("No Object Touch", "Disabled - collisions restored.", 3)
            return
        end

        notify("No Object Touch", "ACTIVE - only your Blobman & WoodPlank can touch you.", 4)

        _noTouchConn = R.Heartbeat:Connect(function()
            if not _noTouchActive then return end
            local char = Player.Character
            if not char then return end

            -- Build whitelist each frame (fast table lookup)
            local whitelist = _getMyWhitelistedModels()

            -- 1) Set ALL character parts to no-touch / no-query
            for _, p in pairs(char:GetDescendants()) do
                if p:IsA("BasePart") then
                    p.CanTouch = false
                    p.CanQuery = false
                end
            end

            -- 2) Destroy incoming welds on character parts that are NOT from whitelist
            for _, p in pairs(char:GetDescendants()) do
                if p:IsA("BasePart") then
                    for _, child in pairs(p:GetChildren()) do
                        if child:IsA("WeldConstraint") or child:IsA("RigidConstraint")
                           or child:IsA("AlignPosition") or child:IsA("AlignOrientation") then
                            local ok0, part0 = pcall(function() return child.Part0 end)
                            local ok1, part1 = pcall(function() return child.Part1 end)
                            local fromWhitelist = false
                            if ok0 and part0 and _isPartInWhitelist(part0, whitelist) then fromWhitelist = true end
                            if ok1 and part1 and _isPartInWhitelist(part1, whitelist) then fromWhitelist = true end
                            -- Also allow self-welds (char to char)
                            if ok0 and part0 and char:IsAncestorOf(part0) then fromWhitelist = true end
                            if ok1 and part1 and char:IsAncestorOf(part1) then fromWhitelist = true end
                            if not fromWhitelist then
                                pcall(function() child:Destroy() end)
                            end
                        end
                    end
                end
            end

            -- 3) Watch for newly-added parts near player and nullify them (Touched-based attacks)
            -- Non-anchored non-whitelisted parts within 12 studs: disable CanTouch on them too
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if hrp then
                local nearby = workspace:GetPartBoundsInBox(hrp.CFrame, Vector3.new(24, 24, 24))
                for _, part in ipairs(nearby) do
                    if part and part.Parent and not part.Anchored then
                        if not char:IsAncestorOf(part) and not _isPartInWhitelist(part, whitelist) then
                            -- Only disable CanTouch on foreign non-anchored parts (does not break map)
                            if part.CanTouch then
                                pcall(function() part.CanTouch = false end)
                            end
                        end
                    end
                end
            end
        end)

        -- Also watch for new welds added to character
        Player.CharacterAdded:Connect(function(newChar)
            if not _noTouchActive then return end
            task.wait(1)
            -- Re-apply on respawn
            if _noTouchConn then _noTouchConn:Disconnect(); _noTouchConn = nil end
            SetToggleState("NoObjectTouch", false)
            SetToggleState("NoObjectTouch", true)
            -- Re-trigger callback by setting back
            _noTouchActive = true
        end)

        _defenseRestarters["NoObjectTouch"] = function()
            if _noTouchActive and not _noTouchConn then
                -- Reconnect after respawn handled by CharacterAdded above
            end
        end
    end
})
